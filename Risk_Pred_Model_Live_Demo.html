import zipfile

class CreditRiskModel:
    """
    Complete Credit Risk Prediction Model with SHAP Explanations
    """

    def __init__(self, random_state=42):
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.explainer = None
        self.feature_names = None
        self.X_train = None
        self.X_val = None
        self.y_train = None
        self.y_val = None

    def load_data(self, data_path=None):
        """
        Load and preprocess the "Give Me Some Credit" dataset
        """
        print("Loading data...")

        # Load dataset zipfile from url
        import requests, zipfile, io
        url = 'https://drive.google.com/uc?export=download&id=1qdZhGZTr80NpdAuEMm9ANIHD8eJYU7tL'
        r = requests.get(url)


        # Unzip the file
        with zipfile.ZipFile(io.BytesIO(r.content), 'r') as zip_ref:
            zip_ref.extractall('/content/credit_data')

        if data_path is None:
            train_url = '/content/credit_data/cs-training.csv'
            self.df = pd.read_csv(train_url, index_col=0)
        else:
            self.df = pd.read_csv(data_path)

        # Rename columns
        self.df.columns = [
            'target', 'revolving_util', 'age', 'delinq_30_59',
            'debt_ratio', 'monthly_income', 'num_open_credit',
            'delinq_90', 'num_real_estate', 'delinq_60_89', 'num_dependents'
        ]

        print(f"Data shape: {self.df.shape}")
        print(f"Default rate: {self.df['target'].mean()*100:.2f}%")

        return self.df

    def preprocess_data(self, df=None):
        """
        Preprocess the data with the same pipeline used in training
        """
        if df is None:
            df = self.df.copy()
        else:
            df = df.copy()

        # 1. Handle outliers in delinquency variables (values 96/98)
        delinquency_cols = ['delinq_30_59', 'delinq_60_89', 'delinq_90']
        for col in delinquency_cols:
            if col in df.columns:
                outlier_mask = (df[col] >= 96)
                if outlier_mask.sum() > 0:
                    median_val = df[~outlier_mask][col].median()
                    df.loc[outlier_mask, col] = median_val

        # 2. Winsorize continuous variables
        continuous_cols = ['revolving_util', 'debt_ratio', 'monthly_income']
        for col in continuous_cols:
            if col in df.columns:
                upper_limit = df[col].quantile(0.99)
                df[col] = df[col].clip(upper=upper_limit)

        # 3. Handle missing values
        if 'monthly_income' in df.columns:
            df['monthly_income'].fillna(df['monthly_income'].median(), inplace=True)
        if 'num_dependents' in df.columns:
            df['num_dependents'].fillna(df['num_dependents'].mode()[0], inplace=True)

        # 4. Feature engineering
        df['income_to_debt'] = df['monthly_income'] / (df['debt_ratio'] + 0.001)
        df['total_delinquencies'] = df['delinq_30_59'] + df['delinq_60_89'] + df['delinq_90']
        df['has_delinquency'] = (df['total_delinquencies'] > 0).astype(int)

        # 5. Remove target if present
        if 'target' in df.columns:
            y = df['target']
            X = df.drop('target', axis=1)
        else:
            y = None
            X = df

        return X, y

    def train_model(self, test_size=0.2, scale_pos_weight=None,
                   n_estimators=200, max_depth=4, learning_rate=0.1):
        """
        Train the XGBoost model
        """
        print("\n" + "="*60)
        print("TRAINING XGBOOST MODEL")
        print("="*60)

        # Prepare data
        if self.X_train is None:
            X, y = self.preprocess_data()
            self.feature_names = X.columns.tolist()

            # Split data
            self.X_train, self.X_val, self.y_train, self.y_val = train_test_split(
                X, y, test_size=test_size, random_state=self.random_state, stratify=y
            )
            print(f"Training set: {self.X_train.shape[0]} samples")
            print(f"Validation set: {self.X_val.shape[0]} samples")

        # Scale features
        self.scaler = StandardScaler()
        X_train_scaled = self.scaler.fit_transform(self.X_train)
        X_val_scaled = self.scaler.transform(self.X_val)

        # Calculate scale_pos_weight if not provided
        if scale_pos_weight is None:
            scale_pos_weight = len(self.y_train[self.y_train==0]) / len(self.y_train[self.y_train==1])
            print(f"Scale_pos_weight: {scale_pos_weight:.2f}")

        # Initialize and train XGBoost
        self.model = xgb.XGBClassifier(
            n_estimators=n_estimators,
            max_depth=max_depth,
            learning_rate=learning_rate,
            scale_pos_weight=scale_pos_weight,
            random_state=self.random_state,
            eval_metric='auc',
            use_label_encoder=False,
            enable_categorical=False # Explicitly disable categorical feature handling
        )

        print("\nTraining XGBoost...")
        self.model.fit(
            X_train_scaled, self.y_train,
            eval_set=[(X_train_scaled, self.y_train), (X_val_scaled, self.y_val)],
            verbose=False
        )

        # Evaluate model
        self._evaluate_model(X_train_scaled, X_val_scaled)

        # Initialize SHAP explainer
        self._init_shap_explainer()

        print("\n✅ Model training complete!")
        return self.model

    def _evaluate_model(self, X_train_scaled, X_val_scaled):
        """
        Evaluate model performance
        """
        # Predictions
        y_pred_train = self.model.predict(X_train_scaled)
        y_pred_val = self.model.predict(X_val_scaled)
        y_proba_train = self.model.predict_proba(X_train_scaled)[:, 1]
        y_proba_val = self.model.predict_proba(X_val_scaled)[:, 1]

        # Metrics
        metrics = {
            'train_auc': roc_auc_score(self.y_train, y_proba_train),
            'val_auc': roc_auc_score(self.y_val, y_proba_val),
            'train_accuracy': accuracy_score(self.y_train, y_pred_train),
            'val_accuracy': accuracy_score(self.y_val, y_pred_val),
            'train_precision': precision_score(self.y_train, y_pred_train, zero_division=0),
            'val_precision': precision_score(self.y_val, y_pred_val, zero_division=0),
            'train_recall': recall_score(self.y_train, y_pred_train, zero_division=0),
            'val_recall': recall_score(self.y_val, y_pred_val, zero_division=0),
            'train_f1': f1_score(self.y_train, y_pred_train, zero_division=0),
            'val_f1': f1_score(self.y_val, y_pred_val, zero_division=0)
        }

        print("\n" + "-"*40)
        print("MODEL PERFORMANCE")
        print("-"*40)
        print(f"Training AUC:  {metrics['train_auc']:.4f}")
        print(f"Validation AUC: {metrics['val_auc']:.4f}")
        print(f"Training F1:   {metrics['train_f1']:.4f}")
        print(f"Validation F1:  {metrics['val_f1']:.4f}")
        print(f"Training Recall: {metrics['train_recall']:.4f}")
        print(f"Validation Recall: {metrics['val_recall']:.4f}")

        self.metrics = metrics

    def _init_shap_explainer(self):
        """
        Initialize SHAP explainer for the model
        """
        print("\nInitializing SHAP explainer...")
        # Use a subset of training data for background (faster)
        background_data = self.X_train[:100]
        background_scaled = self.scaler.transform(background_data)

        self.explainer = shap.TreeExplainer(self.model, background_scaled, feature_perturbation='tree_path_dependent')
        print("✅ SHAP explainer ready!")

    def predict(self, X_new, threshold=0.5):
        """
        Make predictions on new data
        """
        # Preprocess if needed
        if isinstance(X_new, pd.DataFrame):
            X_processed, _ = self.preprocess_data(X_new)
            X_scaled = self.scaler.transform(X_processed)
        else:
            X_scaled = self.scaler.transform(X_new)

        # Make predictions
        predictions = self.model.predict(X_scaled)
        probabilities = self.model.predict_proba(X_scaled)[:, 1]

        # Apply threshold
        final_predictions = (probabilities >= threshold).astype(int)

        return {
            'predictions': final_predictions,
            'probabilities': probabilities,
            'threshold': threshold
        }

    def explain_prediction(self, X_instance, threshold=0.5, visualize=True):
        """
        Get SHAP explanation for a single prediction
        """
        # Preprocess if needed
        if isinstance(X_instance, pd.DataFrame):
            X_processed, _ = self.preprocess_data(X_instance)
            X_scaled = self.scaler.transform(X_processed)
        else:
            X_scaled = self.scaler.transform(X_instance.reshape(1, -1))

        # Calculate SHAP values
        shap_values = self.explainer.shap_values(X_scaled)
        expected_value = self.explainer.expected_value

        # Get prediction
        probability = self.model.predict_proba(X_scaled)[0, 1]
        prediction = (probability >= threshold).astype(int)

        # Create explanation dictionary
        feature_names = self.feature_names
        shap_df = pd.DataFrame({
            'feature': feature_names,
            'shap_value': shap_values[0],
            'feature_value': X_scaled[0]
        })
        shap_df['abs_shap'] = np.abs(shap_df['shap_value'])
        shap_df = shap_df.sort_values('abs_shap', ascending=False)

        # Calculate contribution
        shap_df['contribution'] = shap_df['shap_value'] / np.sum(shap_df['abs_shap'])

        explanation = {
            'prediction': int(prediction),
            'probability': float(probability),
            'threshold': threshold,
            'expected_value': float(expected_value),
            'shap_values': shap_df.to_dict('records'),
            'top_features': shap_df.head(5).to_dict('records'),
            'feature_importance': shap_df[['feature', 'abs_shap']].to_dict('records')
        }

        # Visualize if requested
        if visualize:
            self._visualize_explanation(shap_values, X_scaled[0], expected_value, feature_names)

        return explanation

    def _visualize_explanation(self, shap_values, X_instance, expected_value, feature_names):
        """
        Create visualization for SHAP explanation
        """
        # Waterfall plot
        plt.figure(figsize=(12, 6))
        shap.waterfall_plot(
            shap.Explanation(
                values=shap_values[0],
                base_values=expected_value,
                data=X_instance,
                feature_names=feature_names
            ),
            show=False
        )
        plt.title('SHAP Explanation - Credit Risk Prediction', fontsize=14)
        plt.tight_layout()
        plt.savefig('shap_explanation_waterfall.png', dpi=300, bbox_inches='tight')
        plt.show()

        # Force plot
        plt.figure(figsize=(14, 4))
        shap.force_plot(
            expected_value,
            shap_values[0],
            X_instance,
            feature_names=feature_names,
            matplotlib=True,
            show=False
        )
        plt.title('SHAP Force Plot', fontsize=14)
        plt.tight_layout()
        plt.savefig('shap_explanation_force.png', dpi=300, bbox_inches='tight')
        plt.show()

    def get_feature_importance(self):
        """
        Get global feature importance from the model
        """
        importance = pd.DataFrame({
            'feature': self.feature_names,
            'importance': self.model.feature_importances_
        }).sort_values('importance', ascending=False)

        # Add SHAP importance for comparison
        if self.explainer is not None:
            shap_importance = np.abs(self.explainer.shap_values(self.scaler.transform(self.X_train[:100]))).mean(axis=0)
            importance['shap_importance'] = shap_importance / np.sum(shap_importance)

        return importance

    def save_model(self, filepath='credit_risk_model'):
        """
        Save the trained model and artifacts
        """
        os.makedirs(filepath, exist_ok=True)

        # Save model
        self.model.save_model(f'{filepath}/xgboost_model.json')

        # Save scaler
        with open(f'{filepath}/scaler.pkl', 'wb') as f:
            pickle.dump(self.scaler, f)

        # Save feature names
        with open(f'{filepath}/feature_names.json', 'w') as f:
            json.dump(self.feature_names, f)

        # Save SHAP explainer
        with open(f'{filepath}/explainer.pkl', 'wb') as f:
            pickle.dump(self.explainer, f)

        print(f"\n✅ Model saved to '{filepath}'")

    def load_model(self, filepath='credit_risk_model'):
        """
        Load a saved model and artifacts
        """
        # Load model
        self.model = xgb.XGBClassifier()
        self.model.load_model(f'{filepath}/xgboost_model.json')

        # Load scaler
        with open(f'{filepath}/scaler.pkl', 'rb') as f:
            self.scaler = pickle.load(f)

        # Load feature names
        with open(f'{filepath}/feature_names.json', 'r') as f:
            self.feature_names = json.load(f)

        # Load SHAP explainer
        with open(f'{filepath}/explainer.pkl', 'rb') as f:
            self.explainer = pickle.load(f)

        print(f"\n✅ Model loaded from '{filepath}'")
        return self

class BatchPredictor:
    """
    Handle batch predictions with explanations
    """

    def __init__(self, model):
        self.model = model

    def predict_batch(self, X_batch, threshold=0.5, top_k=5):
        """
        Make predictions for a batch of customers with explanations
        """
        results = []

        for idx in range(len(X_batch)):
            instance = X_batch.iloc[idx:idx+1] if isinstance(X_batch, pd.DataFrame) else X_batch[idx:idx+1]
            explanation = self.model.explain_prediction(instance, threshold=threshold, visualize=False)

            # Format for output
            result = {
                'customer_id': idx,
                'prediction': explanation['prediction'],
                'probability': explanation['probability'],
                'risk_level': self._get_risk_level(explanation['probability']),
                'top_risk_factors': explanation['top_features'][:top_k]
            }
            results.append(result)

        return results

    def _get_risk_level(self, probability):
        """Categorize risk level based on probability"""
        if probability < 0.3:
            return 'Low Risk'
        elif probability < 0.5:
            return 'Medium Risk'
        elif probability < 0.7:
            return 'High Risk'
        else:
            return 'Very High Risk'

    def export_results(self, results, filename='predictions_with_explanations.csv'):
        """
        Export batch results to CSV
        """
        export_data = []
        for result in results:
            row = {
                'customer_id': result['customer_id'],
                'prediction': result['prediction'],
                'probability': result['probability'],
                'risk_level': result['risk_level']
            }
            # Add top risk factors
            for i, factor in enumerate(result['top_risk_factors']):
                row[f"risk_factor_{i+1}"] = f"{factor['feature']}: {factor['shap_value']:.3f}"
            export_data.append(row)

        df = pd.DataFrame(export_data)
        df.to_csv(filename, index=False)
        print(f"\n✅ Results exported to {filename}")
        return df

def create_api_app(model):
    """
    Create a Flask API endpoint for real-time predictions
    (Optional - requires Flask installation)
    """
    try:
        from flask import Flask, request, jsonify

        app = Flask(__name__)

        @app.route('/health', methods=['GET'])
        def health_check():
            return jsonify({'status': 'healthy', 'timestamp': datetime.now().isoformat()})

        @app.route('/predict', methods=['POST'])
        def predict():
            try:
                # Get input data
                data = request.get_json()

                # Convert to DataFrame
                input_data = pd.DataFrame([data['features']])

                # Make prediction with explanation
                threshold = data.get('threshold', 0.5)
                explanation = model.explain_prediction(input_data, threshold=threshold, visualize=False)

                # Format response
                response = {
                    'success': True,
                    'prediction': explanation['prediction'],
                    'probability': explanation['probability'],
                    'risk_level': 'High' if explanation['prediction'] == 1 else 'Low',
                    'top_risk_factors': [
                        {
                            'feature': f['feature'],
                            'impact': f['shap_value'],
                            'contribution': f['contribution']
                        }
                        for f in explanation['top_features'][:5]
                    ],
                    'explanation': {
                        'expected_value': explanation['expected_value'],
                        'feature_values': explanation['shap_values']
                    },
                    'timestamp': datetime.now().isoformat()
                }

                return jsonify(response)

            except Exception as e:
                return jsonify({
                    'success': False,
                    'error': str(e)
                }), 400

        return app

    except ImportError:
        print("Flask not installed. Install with: pip install flask")
        return None
